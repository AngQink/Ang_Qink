cmake_minimum_required(VERSION 3.18)
project(receipt LANGUAGES CXX)

set(BENCH_CHANNEL "Stable" CACHE STRING "Receipt channel")
if(BENCH_CHANNEL STREQUAL "Stable")
  set(RECEIPT_ABI 29)
  set(RECEIPT_STRIDE 19)
  set(RECEIPT_CLAMP 1)
elseif(BENCH_CHANNEL STREQUAL "Canary")
  set(RECEIPT_ABI 29)
  set(RECEIPT_STRIDE 31)
  set(RECEIPT_CLAMP 0)
else()
  message(FATAL_ERROR "Unknown BENCH_CHANNEL")
endif()
configure_file(cmake/contract.hpp.in generated/contract.hpp @ONLY)

add_library(receipt_policy STATIC src/receipt.cpp)
target_compile_features(receipt_policy PUBLIC cxx_std_17)
target_include_directories(receipt_policy PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(receipt src/main.cpp)
target_link_libraries(receipt PRIVATE receipt_policy)
